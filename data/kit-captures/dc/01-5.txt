// Microsoft's emergency access sign-in query (Manage emergency access admin accounts), with the accounts'
// object IDs from the EmergencyAccessIds parameter instead of literals. The alert rule is the control.
let ids = dynamic([{EmergencyAccessIds}]);
SigninLogs
| where TimeGenerated > ago(1d)
| where UserId in (ids)
| project TimeGenerated, UserPrincipalName, UserId, IPAddress, ResultType, ResultDescription
| order by TimeGenerated desc
